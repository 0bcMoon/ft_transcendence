START AIController
  Every reactionMs milliseconds:
    1. state ← getState()

    2. IF state.status ≠ "playing" THEN
         RETURN  // do nothing if game not active

    3. paddleCenter ← paddle.y + paddle.height / 2

    4. // Optional power-up targeting
       targetY ← choosePowerupTarget(state)
       IF targetY ≠ null THEN
            up   ← (paddleCenter > targetY + deadzone)
            down ← (paddleCenter < targetY - deadzone)
            updateIntent(up, down)
            CONTINUE to next loop

    5. // Predict intercept point of ball
       predictedY ← predictBallImpactY(state, side)

    6. // Add human-like imperfection
       targetY ← predictedY + random(-aimError, aimError)

    7. // If ball moving away, drift toward vertical center
       IF ballMovingAwayFromAI(state, side) THEN
            targetY ← lerp(targetY, GAME_HEIGHT/2, centerBias)

    8. // Decide key presses
       up   ← (paddleCenter > targetY + deadzone)
       down ← (paddleCenter < targetY - deadzone)

    9. updateIntent(up, down)
END LOOP


FUNCTION predictBallImpactY(state, side):
    ballX, ballY, vX, vY ← state.ball position and velocity
    paddleX ← AI paddle x position

    IF ball is moving away from AI:
        RETURN GAME_HEIGHT / 2

    // step ball forward until it reaches paddle x
    REPEAT
        ballX ← ballX + vX
        ballY ← ballY + vY

        IF ball hits top/bottom wall:
            vY ← -vY
    UNTIL ball crosses paddleX OR max iterations reached

    RETURN clamp(ballY, 0, GAME_HEIGHT)
END FUNCTION

FUNCTION choosePowerupTarget(state):
    IF powerupAggro == 0 OR no active power-ups:
        RETURN null

    candidates ← active power-ups on AI’s side, sorted by proximity in x

    IF candidates empty OR random() > powerupAggro:
        RETURN null

    RETURN candidates[0].y
END FUNCTION
